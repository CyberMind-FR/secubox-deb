# tests/publiques-assumees.txt — routes publiques A DESSEIN (#1256)
#
# CE FICHIER N'EST PAS UNE DEROGATION DE CONFORT. Une route n'y entre que si
# elle porte SA PROPRE protection, ou si exiger un jeton la casserait par
# construction. Chaque ligne dit laquelle des deux, en clair — sans raison
# verifiable, la route retourne dans tests/dette-jwt.txt.
#
# Format : <module> <METHODE> <chemin>   # raison
#
# Les entrees marquees « A CONFIRMER » reposent sur une lecture du code, pas
# sur un essai sur la board : elles sont a valider par l'operateur.

# ── Points d'entree d'authentification : ils DELIVRENT le jeton ────────────
secubox-auth POST /login                # delivre le jeton — ne peut pas en exiger un
secubox-auth POST /login/mfa            # 2e facteur, porte par le mfa_token du 1er
secubox-auth POST /totp/enroll          # porte par l'enrollment_token
secubox-auth POST /totp/confirm         # porte par l'enrollment_token
secubox-auth POST /set-password         # porte par le setup_token
secubox-auth POST /redeem_voucher       # documente « public endpoint » dans le code

# ── Session propre : cookie + CSRF double-envoi, pas de JWT ────────────────
# secubox-billets a son admin rendu cote serveur : _current_author() lit le
# cookie billets_session et sec.csrf_ok() verifie le double-envoi. Poser
# require_jwt casserait la connexion admin sans rien ajouter.
secubox-billets POST /admin/login                       # delivre la session
secubox-billets POST /admin/logout                      # ferme la session
secubox-billets POST /admin/override                    # session + CSRF
secubox-billets POST /admin/password                    # session + CSRF
secubox-billets POST /admin/billets                     # session + CSRF
secubox-billets POST /admin/billets/{billet_id}         # session + CSRF
secubox-billets POST /admin/billets/{billet_id}/refetch # session + CSRF
secubox-billets POST /admin/billets/{billet_id}/delete  # session + CSRF
secubox-billets POST /admin/media/{media_id}/delete     # session + CSRF
secubox-billets POST /admin/comments/{comment_id}/moderate # session + CSRF
secubox-billets POST /b/{slug}/react                    # blog public, par conception
secubox-billets POST /b/{slug}/comment                  # blog public, par conception

# ── Protocoles machine a machine, avec leur propre preuve ─────────────────
secubox-metablogizer POST /webhook      # webhook Gitea verifie en HMAC
secubox-soc-gateway POST /enroll        # enrolement d'un noeud edge, requete signee
secubox-soc-gateway POST /ingest        # requete signee (documente dans le code)
secubox-soc-gateway POST /regional/enroll # enrolement d'un SOC regional
secubox-soc-gateway POST /regional/ingest # requete signee
secubox-soc-agent POST /command         # exige node_token + signature verifiee par execute_command()
secubox-threatmesh POST /mesh/ingest    # decisions d'un pair du mesh, transport authentifie
secubox-p2p POST /master-link/token/validate  # documente « public » : valide un jeton, n'en exige pas
secubox-p2p POST /master-link/join      # jeton valide dans le corps de la requete
secubox-p2p POST /master-link/invite-local    # restreint a localhost (documente)
secubox-p2p POST /master-link/cleanup-local   # restreint a localhost (documente)
secubox-p2p POST /api/v1/p2p-macro/grant/{service_id} # Subscription auto-signee, verifiee
secubox-mail POST /autodiscover/autodiscover.xml  # protocole Outlook : le client ne peut pas porter de JWT
secubox-mail POST /Autodiscover/Autodiscover.xml  # idem, variante de casse imposee par le protocole
secubox-droplet POST /depot             # documente « PUBLIC — c'est la raison d'etre de cet espace »
secubox-eye-remote POST /api/v1/eye-remote/connected     # appele par udev, en local
secubox-eye-remote POST /api/v1/eye-remote/disconnected  # appele par udev, en local
